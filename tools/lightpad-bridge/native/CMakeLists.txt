cmake_minimum_required(VERSION 3.16)
project(lightpad_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(protocol_test protocol_test.cpp)
target_compile_options(protocol_test PRIVATE -UNDEBUG)
add_test(NAME protocol COMMAND protocol_test)
set(BLOCKS_SDK_DIR "" CACHE PATH "ROLI BLOCKS SDK module directory")
if(NOT APPLE)
  message(FATAL_ERROR "The bridge currently requires macOS")
endif()
if(NOT EXISTS "${BLOCKS_SDK_DIR}/roli_blocks_basics/roli_blocks_basics.cpp")
  message(FATAL_ERROR "Set BLOCKS_SDK_DIR to the local SDK module directory")
endif()
# Internal connection is inline and included unchanged to satisfy three unsafe
# vendor static_casts. Fail closed if the pinned definition changes.
file(SHA256 "${BLOCKS_SDK_DIR}/roli_blocks_basics/topology/internal/roli_MidiDeviceConnection.cpp" CONNECTION_SHA)
if(NOT CONNECTION_SHA STREQUAL "6c01825a8eebd2588fd3f01bb049c8712e167511eef424b6acdcf098b58774ce")
  message(FATAL_ERROR "SDK internal MIDI connection changed; re-audit its layout before building")
endif()
enable_language(OBJCXX)
set(CMAKE_OBJCXX_STANDARD 17)
set(SDK_SOURCES "${BLOCKS_SDK_DIR}/roli_blocks_basics/roli_blocks_basics.cpp")
foreach(module juce_core juce_events juce_audio_basics juce_audio_devices)
  list(APPEND SDK_SOURCES "${BLOCKS_SDK_DIR}/${module}/${module}.mm")
endforeach()
add_library(bridge_sdk STATIC ${SDK_SOURCES})
target_include_directories(bridge_sdk SYSTEM PUBLIC "${BLOCKS_SDK_DIR}")
target_compile_definitions(bridge_sdk PUBLIC JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_DONT_DECLARE_PROJECTINFO=1 JUCE_STANDALONE_APPLICATION=1 JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0 JucePlugin_CFBundleIdentifier="net.xiphe.lightpad-bridge" $<$<CONFIG:Debug>:DEBUG=1;_DEBUG=1> $<$<NOT:$<CONFIG:Debug>>:NDEBUG=1>)
target_compile_options(bridge_sdk PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fno-objc-arc>)
foreach(framework Cocoa Foundation IOKit CoreAudio CoreMIDI AudioToolbox Accelerate)
  find_library(FRAMEWORK_${framework} ${framework} REQUIRED)
  target_link_libraries(bridge_sdk PUBLIC "${FRAMEWORK_${framework}}")
endforeach()
add_executable(lightpad-bridge main.cpp event_loop.mm)
target_compile_options(lightpad-bridge PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fno-objc-arc>)
target_link_libraries(lightpad-bridge PRIVATE bridge_sdk)
target_compile_options(lightpad-bridge PRIVATE -Wall -Wextra -Wpedantic)
add_executable(policy_test policy_test.cpp)
target_link_libraries(policy_test PRIVATE bridge_sdk)
add_test(NAME policy COMMAND policy_test)

add_executable(modal_test modal_test.cpp)
target_compile_options(modal_test PRIVATE -UNDEBUG)
add_test(NAME modal COMMAND modal_test)

add_executable(short_long_test short_long_test.cpp)
target_compile_options(short_long_test PRIVATE -UNDEBUG)
add_test(NAME short_long COMMAND short_long_test)
